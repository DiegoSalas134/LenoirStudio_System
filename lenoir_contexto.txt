import express, { Router } from "express";
import helmet from "helmet";
import cookieParser from "cookie-parser";
import cors from "cors";
import { AppError, errorHandler } from "./middlewares/errorHandler.ts";
import path from "node:path";

interface ServerOptions {
  port: number;
  routes: Router;
}

export class Server {
  readonly app = express();

  constructor(private readonly options: ServerOptions) {
    this.app.use(helmet());

    this.app.use(cors({
      origin: ["http://127.0.0.1:5500", "http://localhost:5500"],
      credentials: true
    }));
    
    this.app.use(express.json({ limit: "100kb" }));
    this.app.use(cookieParser());
    this.app.use(express.static(path.resolve(import.meta.dirname, "../../frontend/test")));
    this.app.use("/api/v1", options.routes);
    this.app.use((_req, _res, next) => next(new AppError(404, "NOT_FOUND", "Route not found")));
    this.app.use(errorHandler);
  }

  start() {
    this.app.listen(this.options.port, () => {
      console.log(`server running on port: ${this.options.port}`);
    });
  }
}

import dotenv from "dotenv";
dotenv.config();
import { Server } from "./server.js";
import { AppRoutes } from "./routes/index.js";
import { pool } from "./config/database.js";
const port = Number(process.env["PORT"]) || 3000;
async function main() {
    try {
        const connection = await pool.getConnection();
        await connection.query("SELECT 1");
        connection.release();
        console.log("MySQL conectado correctamente.");
        const server = new Server({
            port,
            routes: AppRoutes.router
        });
        server.start();
    }
    catch (error) {
        console.error("Error al conectar con MySQL:", error);
        await pool.end();
        process.exitCode = 1;
    }
}
main();

import express, { Router } from "express";
import cors from "cors";
export class Server {
    server = express();
    port;
    constructor(options) {
        this.port = options.port;
        this.server.use(cors({
            origin: "http://127.0.0.1:5500"
        }));
        this.server.use(express.json());
        this.server.use('/api/v1', options.routes);
    }
    start() {

}

import { Router } from "express";
import { AuthController } from "../controllers/auth.controller.ts";
import { authenticate, requireAdmin } from "../middlewares/authenticate.ts";
import { loginRateLimiter } from "../middlewares/rateLimit.ts";
import { validate } from "../middlewares/validate.ts";
import { loginValidator } from "../validators/auth.validator.ts";

export class AuthRoutes {
  static get router(): Router {
    const router = Router();
    router.post("/login", loginRateLimiter, loginValidator, validate, AuthController.login);
    router.post("/logout", authenticate, requireAdmin, AuthController.logout);
    router.get("/me", authenticate, requireAdmin, AuthController.me);
    return router;
  }
}

import { Router } from "express";
import { QuoteController } from "../controllers/quote.controller.ts";
import { publicWriteLimiter } from "../middlewares/rateLimit.ts";
import { parseQuoteField, upload } from "../middlewares/upload.ts";
import { validate } from "../middlewares/validate.ts";
import { quoteCodeValidator, quoteValidator } from "../validators/quote.validator.ts";

export class QuoteRoutes {
  static get router(): Router {
    const router = Router();
    router.get("/catalog", QuoteController.catalog);
    router.post("/quotes/calculate", quoteValidator, validate, QuoteController.calculate);
    router.post(
      "/quotes",
      publicWriteLimiter,
      upload.single("referencePhoto"),
      parseQuoteField,
      quoteValidator,
      validate,
      QuoteController.save,
    );
    return router;
  }

  static get adminRouter(): Router {
    const router = Router();
    router.get("/", QuoteController.list);
    router.get("/:code/photo", quoteCodeValidator, validate, QuoteController.photo);
    return router;
  }
}

import { Router } from "express";
import { AppointmentController } from "../controllers/appointment.controller.ts";
import { publicWriteLimiter } from "../middlewares/rateLimit.ts";
import { upload } from "../middlewares/upload.ts";
import { validate } from "../middlewares/validate.ts";
import {
  appointmentCodeValidator,
  appointmentListValidator,
  appointmentValidator,
  availabilityValidator,
  confirmValidator,
} from "../validators/appointment.validator.ts";

export class AppointmentRoutes {
  static get router(): Router {
    const router = Router();
    router.get("/availability", availabilityValidator, validate, AppointmentController.availability);
    router.post(
      "/appointments",
      publicWriteLimiter,
      upload.single("receipt"),
      appointmentValidator,
      validate,
      AppointmentController.create,
    );
    return router;
  }
  static get adminRouter(): Router {
    const router = Router();
    router.get("/", appointmentListValidator, validate, AppointmentController.list);
    router.get("/:code/receipt", appointmentCodeValidator, validate, AppointmentController.receipt);
    router.patch("/:code/confirm", confirmValidator, validate, AppointmentController.confirm);
    router.patch("/:code/cancel", appointmentCodeValidator, validate, AppointmentController.cancel);
    return router;
  }
}

import { Router } from "express";
import { AppointmentRoutes } from "./appointment.routes.ts";
import { AuthRoutes } from "./auth.routes.ts";
import { QuoteRoutes } from "./quote.routes.ts";
import { authenticate, noStore, requireAdmin } from "../middlewares/authenticate.ts";

export class AppRoutes {
  static get router(): Router {
    const router = Router();
    router.get("/health", (_req, res) => {
      res.status(200).json({ status: "ok" });
    });
    router.use("/admin", noStore);
    router.use("/admin/auth", AuthRoutes.router);
    router.use("/admin/quotes", authenticate, requireAdmin, QuoteRoutes.adminRouter);
    router.use("/admin/appointments", authenticate, requireAdmin, AppointmentRoutes.adminRouter);
    router.use("/", QuoteRoutes.router);
    router.use("/", AppointmentRoutes.router);
    return router;
  }
}
